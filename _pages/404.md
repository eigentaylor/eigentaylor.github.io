You will be redirected within 3 seconds. If not redirected, please go back to the [home page]({{ site.baseurl | prepend: site.url }}).

<script>
(function () {
  var path = window.location.pathname;
  var previewMatch = path.match(/^\/preview\/([^\/]+)\/?$/);

  function redirectHome() {
    window.location.replace('/');
  }

  if (previewMatch) {
    var slug = previewMatch[1];
    var blogUrl = '/blog/' + slug + '/';
    fetch(blogUrl, { method: 'HEAD' })
      .then(function (resp) {
        if (resp.ok) {
          window.location.replace(blogUrl);
        } else {
          setTimeout(redirectHome, 3000);
        }
      })
      .catch(function () {
        setTimeout(redirectHome, 3000);
      });
  } else {
    setTimeout(redirectHome, 3000);
  }
})();
</script>
